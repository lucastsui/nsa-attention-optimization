384           # 1. lse computation                                                                                                       
385           ################################                                                                                           
386           if lse is not None:                                                                                                        
387               b_lse = tl.load(lse + (bos + i_t) * HQ + i_h * G + tl.arange(0, G))                                                    
388           else:                                                                                                                      
389               # max scores for the current block                                                                                     
390               b_m = tl.full([G], float('-inf'), dtype=tl.float32)                                                                    
391               # lse = log(acc) + m                                                                                                   
392               b_acc = tl.zeros([G], dtype=tl.float32)                                                                                
393               for i_c in range(0, NC, BC):                                                                                           
394                   o_c = i_c + tl.arange(0, BC)                                                                                       
395                                                                                                                                      
396                   p_k = tl.make_block_ptr(k + (boc * H + i_h) * K, (K, TC), (1, H*K), (0, i_c), (BK, BC), (0, 1))                    
397                   # [BK, BC]                                                                                                         
398                   b_k = tl.load(p_k, boundary_check=(0, 1))                                                                          
399                                                                                                                                      
400                   # [G, BC]                                                                                                          
401                   b_s = tl.dot(b_q, b_k)                                                                                             
402                   b_s = tl.where((o_c < NC)[None, :], b_s, float('-inf'))                                                            
403                                                                                                                                      
404                   # [G]                                                                                                              
405                   b_m, b_mp = tl.maximum(b_m, tl.max(b_s, 1)), b_m                                                                   
406                   b_r = tl.exp(b_mp - b_m)                                                                                           
407                   # [G, BC]                                                                                                          
408                   b_p = tl.exp(b_s - b_m[:, None])                                                                                   
409                   # [G]                                                                                                              
410                   b_acc = b_acc * b_r + tl.sum(b_p, 1)                                                                               
411                                                                                                                                      
412                   b_mp = b_m                                                                                                         
413               if NC == 0:                                                                                                            
414                   b_lse = tl.zeros([G], dtype=tl.float32)                                                                            
415               else:                                                                                                                  
416                   b_lse = b_m + tl.log(b_acc)                                                                                        
417                                                                                                                                      
418           ################################                                                                                           
419           # 2. topk selection                                                                                                        
420           ################################                                                                                           
421           # [BC]                                                                                                                     
422           b_i = tl.full([BC], -1, dtype=tl.float32)                                                                                  
423           o_i = tl.zeros([BC], dtype=tl.int32)                                                                                       
424           m_i = tl.arange(0, BC) < BC//2                                                                                             
425           for i_c in range(0, i_t // BS + 1, BC):                                                                                    
426               o_c = i_c + tl.arange(0, BC)                                                                                           
427                                                                                                                                      
428               p_k = tl.make_block_ptr(k + (boc * H + i_h) * K, (K, TC), (1, H*K), (0, i_c), (BK, BC), (0, 1))                        
429               # [BK, BC]                                                                                                             
430               b_k = tl.load(p_k, boundary_check=(0, 1))                                                                              
431               # [G, BC]                                                                                                              
432               b_s = tl.dot(b_q, b_k)                                                                                                 
433               b_s = tl.where((i_t // BS > o_c)[None, :], b_s, float('-inf'))                                                         
434               # [G, BC]                                                                                                              
435               b_p = tl.where((i_t // BS == o_c)[None, :], float(1.0), tl.exp(b_s - b_lse[:, None]))                                  
436               # the importance scores of the current block                                                                           
437               # [BC]                                                                                                                 
438               b_i, b_ip = tl.sum(b_p, 0), b_i                                                                                        
439               o_i, o_ip = tl.where(o_c <= i_t // BS, o_c + 1, 0), o_i                                                                
440                                                                                                                                      
441               n_dims: tl.constexpr = tl.standard._log2(b_i.shape[0])                                                                 
442               for i in tl.static_range(1, n_dims):                                                                                   
443                   b_i, o_i = _bitonic_merge(b_i, o_i.to(tl.int32), i, 2, n_dims)                                                     
444                                                                                                                                      
445               if i_c != 0:                                                                                                           
446                   b_i, o_i = _bitonic_merge(b_i, o_i.to(tl.int32), n_dims, False, n_dims)                                            
447                   b_i_new = b_ip * m_i + b_i * (1 - m_i)                                                                             
448                   o_i_new = o_ip * m_i + o_i * (1 - m_i)                                                                             
449                   b_i, o_i = _bitonic_merge(b_i_new, o_i_new.to(tl.int32), n_dims, True, n_dims)                                     
450               else:                                                                                                                  
451                   b_i, o_i = _bitonic_merge(b_i, o_i.to(tl.int32), n_dims, True, n_dims)                                             
452                                                                                                                                      
453           m_top = tl.arange(0, BC//S) == 0                                                                                           
454           b_top = tl.sum(m_top[:, None] * tl.reshape(o_i - 1, [BC//S, S]), 0)                                                        
455                                                                                                                                      
456           p_b = tl.make_block_ptr(block_indices + (bos + i_t) * H*S, (H*S,), (1,), (i_h * S,), (S,), (0,))                           
457           tl.store(p_b, b_top.to(p_b.dtype.element_ty))

475           v,                                                                                                                         
476           o,                                                                                                                         
477           lse,

486           G: tl.constexpr,

490           BS: tl.constexpr,

492           BV: tl.constexpr,

494           USE_BLOCK_COUNTS: tl.constexpr                                                                                             
495       ):                                                                                                                             
496           i_t, i_v, i_bh = tl.program_id(0), tl.program_id(1), tl.program_id(2)

499           if USE_OFFSETS:                                                                                                            
500               i_n, i_t = tl.load(token_indices + i_t * 2).to(tl.int32), tl.load(token_indices + i_t * 2 + 1).to(tl.int32)            
501               bos, eos = tl.load(offsets + i_n).to(tl.int32), tl.load(offsets + i_n + 1).to(tl.int32)

507           v += (bos * H + i_h) * V

524           b_o = tl.zeros([G, BV], dtype=tl.float32)

527           b_acc = tl.zeros([G], dtype=tl.float32)                                                                                    
528           for i in range(NS):                                                                                                        
529               i_s = tl.load(block_indices + i).to(tl.int32) * BS

532                   p_v = tl.make_block_ptr(v, (T, V), (H*V, 1), (i_s, i_v * BV), (BS, BV), (1, 0))                                    
533                   # [BK, BS]                                                                                                         
534                   b_k = tl.load(p_k, boundary_check=(0, 1))                                                                          
535                   # [BS, BV]                                                                                                         
536                   b_v = tl.load(p_v, boundary_check=(0, 1))                                                                          
537                   # [G, BS]                                                                                                          
538                   b_s = tl.dot(b_q, b_k)                                                                                             
539                   b_s = tl.where((i_t >= (i_s + tl.arange(0, BS)))[None, :], b_s, float('-inf'))                                     
540                                                                                                                                      
541                   # [G]                                                                                                              
542                   b_m, b_mp = tl.maximum(b_m, tl.max(b_s, 1)), b_m                                                                   
543                   b_r = tl.exp(b_mp - b_m)                                                                                           
544                   # [G, BS]                                                                                                          
545                   b_p = tl.exp(b_s - b_m[:, None])                                                                                   
546                   # [G]                                                                                                              
547                   b_acc = b_acc * b_r + tl.sum(b_p, 1)                                                                               
548                   # [G, BV]                                                                                                          
549                   b_o = b_o * b_r[:, None] + tl.dot(b_p.to(b_q.dtype), b_v)                                                          
550                                                                                                                                      
551                   b_mp = b_m                                                                                                         
552           b_o = b_o / b_acc[:, None]                                                                                                 
553           b_m += tl.log(b_acc)                                                                                                       
554           tl.store(p_o, b_o.to(p_o.dtype.element_ty), boundary_check=(0, 1))                                                         
555           tl.store(p_lse, b_m.to(p_lse.dtype.element_ty))

760           p_k = tl.make_block_ptr(k + (bos * H + i_h) * K, (T, K), (H*K, 1), (i_s * BS, 0), (BS, BK), (1, 0))                        
761           p_v = tl.make_block_ptr(v + (bos * H + i_h) * V, (T, V), (H*V, 1), (i_s * BS, i_v * BV), (BS, BV), (1, 0))                 
762           p_dk = tl.make_block_ptr(dk + (i_v * B*T*H + bos * H + i_h) * K, (T, K), (H*K, 1), (i_s * BS, 0), (BS, BK), (1, 0))        
763           p_dv = tl.make_block_ptr(dv + (bos * H + i_h) * V, (T, V), (H*V, 1), (i_s * BS, i_v * BV), (BS, BV), (1, 0))               
764                                                                                                                                      
765           # [BS, BK]                                                                                                                 
766           b_k = tl.load(p_k, boundary_check=(0, 1))                                                                                  
767           b_dk = tl.zeros([BS, BK], dtype=tl.float32)                                                                                
768           # [BS, BV]                                                                                                                 
769           b_v = tl.load(p_v, boundary_check=(0, 1))                                                                                  
770           b_dv = tl.zeros([BS, BV], dtype=tl.float32)                                                                                
771                                                                                                                                      
772           for i in range(i_s * BS, T):                                                                                               
773               b_m = tl.load(block_mask + (bos + i) * H*M + i_h * M + i_s)                                                            
774               if b_m:                                                                                                                
775                   p_q = tl.make_block_ptr(q + (bos + i) * HQ*K, (HQ, K), (K, 1), (i_h * G, 0), (G, BK), (1, 0))                      
776                   # [G, BK]                                                                                                          
777                   b_q = tl.load(p_q, boundary_check=(0, 1))                                                                          
778                   b_q = (b_q * scale).to(b_q.dtype)                                                                                  
779                                                                                                                                      
780                   p_do = tl.make_block_ptr(do + (bos + i) * HQ*V, (HQ, V), (V, 1), (i_h * G, i_v * BV), (G, BV), (1, 0))             
781                   p_lse = lse + (bos + i) * HQ + i_h * G + tl.arange(0, G)                                                           
782                   p_delta = delta + (bos + i) * HQ + i_h * G + tl.arange(0, G)                                                       
783                   # [G, BV]                                                                                                          
784                   b_do = tl.load(p_do, boundary_check=(0, 1))                                                                        
785                   # [G]                                                                                                              
786                   b_lse = tl.load(p_lse)                                                                                             
787                   b_delta = tl.load(p_delta)                                                                                         
788                   # [BS, G]                                                                                                          
789                   b_s = tl.dot(b_k, tl.trans(b_q))                                                                                   
790                   b_p = tl.exp(b_s - b_lse[None, :])                                                                                 
791                   b_p = tl.where((i >= (i_s * BS + tl.arange(0, BS)))[:, None], b_p, 0)                                              
792                   # [BS, G] @ [G, BV] -> [BS, BV]                                                                                    
793                   b_dv += tl.dot(b_p.to(b_do.dtype), b_do)                                                                           
794                   # [BS, BV] @ [BV, G] -> [BS, G]                                                                                    
795                   b_dp = tl.dot(b_v, tl.trans(b_do))                                                                                 
796                   # [BS, G]                                                                                                          
797                   b_ds = b_p * (b_dp - b_delta[None, :])                                                                             
798                   # [BS, G] @ [G, BK] -> [BS, BK]                                                                                    
799                   b_dk += tl.dot(b_ds.to(b_q.dtype), b_q)                                                                            
800                                                                                                                                      
801           tl.store(p_dk, b_dk.to(p_dk.dtype.element_ty), boundary_check=(0, 1))                                                      
802           tl.store(p_dv, b_dv.to(p_dv.dtype.element_ty), boundary_check=(0, 1))                                                      
803

805       def parallel_nsa_compression_fwd(                                                                                              
806           q: torch.Tensor,                                                                                                           
807           k: torch.Tensor,                                                                                                           
808           v: torch.Tensor,                                                                                                           
809           block_size: int,                                                                                                           
810           scale: float,                                                                                                              
811           offsets: Optional[torch.LongTensor] = None,                                                                                
812           token_indices: Optional[torch.LongTensor] = None,                                                                          
813       ):                                                                                                                             
814           B, T, HQ, K, V = *q.shape, v.shape[-1]                                                                                     
815           H = k.shape[2]                                                                                                             
816           G = HQ // H                                                                                                                
817           BC = BS = block_size                                                                                                       
818           if torch.cuda.get_device_capability()[0] >= 9:                                                                             
819               BK = min(256, triton.next_power_of_2(K))                                                                               
820               BV = min(256, triton.next_power_of_2(V))                                                                               
821           else:                                                                                                                      
822               BK = min(128, triton.next_power_of_2(K))                                                                               
823               BV = min(128, triton.next_power_of_2(V))                                                                               
824           NK = triton.cdiv(K, BK)                                                                                                    
825           NV = triton.cdiv(V, BV)                                                                                                    
826           assert NK == 1, "The key dimension can not be larger than 256"                                                             
827                                                                                                                                      
828           chunk_offsets = prepare_chunk_offsets(offsets, BS) if offsets is not None else None                                        
829                                                                                                                                      
830           grid = (T, NV, B * H)                                                                                                      
831           o = torch.empty(B, T, HQ, V, dtype=v.dtype, device=q.device)                                                               
832           lse = torch.empty(B, T, HQ, dtype=torch.float, device=q.device)                                                            
833                                                                                                                                      
834           parallel_nsa_compression_fwd_kernel[grid](                                                                                 
835               q=q,                                                                                                                   
836               k=k,                                                                                                                   
837               v=v,                                                                                                                   
838               o=o,                                                                                                                   
839               lse=lse,                                                                                                               
840               scale=scale,                                                                                                           
841               offsets=offsets,                                                                                                       
842               token_indices=token_indices,                                                                                           
843               chunk_offsets=chunk_offsets,                                                                                           
844               T=T,                                                                                                                   
845               H=H,                                                                                                                   
846               HQ=HQ,                                                                                                                 
847               G=G,                                                                                                                   
848               K=K,                                                                                                                   
849               V=V,                                                                                                                   
850               BC=BC,                                                                                                                 
851               BS=BS,                                                                                                                 
852               BK=BK,                                                                                                                 
853               BV=BV,                                                                                                                 
854           )                                                                                                                          
855           return o, lse

858       def parallel_nsa_compression_bwd(                                                                                              
859           q: torch.Tensor,                                                                                                           
860           k: torch.Tensor,                                                                                                           
861           v: torch.Tensor,                                                                                                           
862           o: torch.Tensor,                                                                                                           
863           lse: torch.Tensor,                                                                                                         
864           do: torch.Tensor,                                                                                                          
865           block_size: int = 64,                                                                                                      
866           scale: float = None,                                                                                                       
867           offsets: Optional[torch.LongTensor] = None,                                                                                
868           token_indices: Optional[torch.LongTensor] = None,                                                                          
869       ):                                                                                                                             
870           B, T, HQ, K, V = *q.shape, v.shape[-1]                                                                                     
871           H = k.shape[2]                                                                                                             
872           G = HQ // H                                                                                                                
873           BC = BS = block_size                                                                                                       
874           BK = triton.next_power_of_2(K)                                                                                             
875           BV = min(128, triton.next_power_of_2(v.shape[-1]))                                                                         
876           NV = triton.cdiv(V, BV)                                                                                                    
877           if offsets is not None:                                                                                                    
878               lens = prepare_lens(offsets)                                                                                           
879               chunk_indices = torch.cat([torch.arange(n) for n in triton.cdiv(triton.cdiv(lens, BS), BC).tolist()])                  
880               chunk_indices = torch.stack([chunk_indices.eq(0).cumsum(0) - 1, chunk_indices], 1).to(offsets)                         
881               chunk_offsets = prepare_chunk_offsets(offsets, BS)                                                                     
882               NC = len(chunk_indices)                                                                                                
883           else:                                                                                                                      
884               chunk_indices, chunk_offsets = None, None                                                                              
885               NC = triton.cdiv(triton.cdiv(T, BS), BC)                                                                               
886                                                                                                                                      
887           delta = parallel_nsa_bwd_preprocess(o, do)                                                                                 
888                                                                                                                                      
889           dq = torch.empty(NV, *q.shape, dtype=q.dtype if NV == 1 else torch.float, device=q.device)                                 
890           grid = (T, NV, B * H)                                                                                                      
891           parallel_nsa_compression_bwd_kernel_dq[grid](                                                                              
892               q=q,                                                                                                                   
893               k=k,                                                                                                                   
894               v=v,                                                                                                                   
895               lse=lse,                                                                                                               
896               delta=delta,                                                                                                           
897               do=do,                                                                                                                 
898               dq=dq,                                                                                                                 
899               scale=scale,                                                                                                           
900               offsets=offsets,                                                                                                       
901               token_indices=token_indices,                                                                                           
902               chunk_offsets=chunk_offsets,                                                                                           
903               T=T,                                                                                                                   
904               B=B,                                                                                                                   
905               H=H,                                                                                                                   
906               HQ=HQ,                                                                                                                 
907               G=G,                                                                                                                   
908               K=K,                                                                                                                   
909               V=V,                                                                                                                   
910               BC=BC,                                                                                                                 
911               BS=BS,                                                                                                                 
912               BK=BK,                                                                                                                 
913               BV=BV                                                                                                                  
914           )                                                                                                                          
915           dq = dq.sum(0)                                                                                                             
916                                                                                                                                      
917           dk = torch.empty(NV, *k.shape, dtype=k.dtype if NV == 1 else torch.float, device=q.device)                                 
918           dv = torch.empty(v.shape, dtype=v.dtype, device=q.device)                                                                  
919                                                                                                                                      
920           grid = (NV, NC, B * H)                                                                                                     
921           parallel_nsa_compression_bwd_kernel_dkv[grid](                                                                             
922               q=q,                                                                                                                   
923               k=k,                                                                                                                   
924               v=v,                                                                                                                   
925               lse=lse,                                                                                                               
926               delta=delta,                                                                                                           
927               do=do,                                                                                                                 
928               dk=dk,                                                                                                                 
929               dv=dv,                                                                                                                 
930               offsets=offsets,                                                                                                       
931               chunk_indices=chunk_indices,                                                                                           
932               chunk_offsets=chunk_offsets,                                                                                           
933               scale=scale,                                                                                                           
934               T=T,                                                                                                                   
935               B=B,                                                                                                                   
936               H=H,                                                                                                                   
937               HQ=HQ,                                                                                                                 
938               G=G,                                                                                                                   
939               K=K,                                                                                                                   
940               V=V,                                                                                                                   
941               BC=BC,                                                                                                                 
942               BS=BS,                                                                                                                 
943               BK=BK,                                                                                                                 
944               BV=BV                                                                                                                  
945           )                                                                                                                          
946           dk = dk.sum(0)                                                                                                             
947           return dq, dk, dv

950       class ParallelNSACompressionFunction(torch.autograd.Function):                                                                 
951                                                                                                                                      
952           @staticmethod                                                                                                              
953           @contiguous                                                                                                                
954           @autocast_custom_fwd                                                                                                       
955           def forward(                                                                                                               
956               ctx,                                                                                                                   
957               q,                                                                                                                     
958               k,                                                                                                                     
959               v,                                                                                                                     
960               block_size,                                                                                                            
961               scale,                                                                                                                 
962               offsets                                                                                                                
963           ):                                                                                                                         
964               ctx.dtype = q.dtype                                                                                                    
965                                                                                                                                      
966               # 2-d sequence indices denoting the offsets of tokens in each sequence                                                 
967               # for example, if the passed `offsets` is [0, 2, 6],                                                                   
968               # then there are 2 and 4 tokens in the 1st and 2nd sequences respectively, and `token_indices` will be                 
969               # [[0, 0], [0, 1], [1, 0], [1, 1], [1, 2], [1, 3]]                                                                     
970               token_indices = prepare_token_indices(offsets) if offsets is not None else None                                        
971                                                                                                                                      
972               o, lse = parallel_nsa_compression_fwd(                                                                                 
973                   q=q,                                                                                                               
974                   k=k,                                                                                                               
975                   v=v,                                                                                                               
976                   block_size=block_size,                                                                                             
977                   scale=scale,                                                                                                       
978                   offsets=offsets,                                                                                                   
979                   token_indices=token_indices                                                                                        
980               )                                                                                                                      
981               ctx.save_for_backward(q, k, v, o, lse)                                                                                 
982               ctx.offsets = offsets                                                                                                  
983               ctx.token_indices = token_indices                                                                                      
984               ctx.block_size = block_size                                                                                            
985               ctx.scale = scale                                                                                                      
986               return o.to(q.dtype), lse                                                                                              
987                                                                                                                                      
988           @staticmethod                                                                                                              
989           @contiguous                                                                                                                
990           @autocast_custom_bwd                                                                                                       
991           def backward(ctx, do, *args):                                                                                              
992               q, k, v, o, lse = ctx.saved_tensors                                                                                    
993               dq, dk, dv = parallel_nsa_compression_bwd(                                                                             
994                   q=q,                                                                                                               
995                   k=k,                                                                                                               
996                   v=v,                                                                                                               
997                   o=o,                                                                                                               
998                   lse=lse,                                                                                                           
999                   do=do,                                                                                                             
1000                  block_size=ctx.block_size,                                                                                         
1001                  scale=ctx.scale,                                                                                                   
1002                  offsets=ctx.offsets,                                                                                               
1003                  token_indices=ctx.token_indices                                                                                    
1004              )                                                                                                                      
1005              return dq.to(q), dk.to(k), dv.to(v), None, None, None

1201                                                                                                                                     
1202          if offsets is not None:                                                                                                    
1203              chunk_indices = prepare_chunk_indices(offsets, BS)                                                                     
1204              NS = len(chunk_indices)                                                                                                
1205          else:                                                                                                                      
1206              chunk_indices = None                                                                                                   
1207              NS = triton.cdiv(T, BS)                                                                                                
1208                                                                                                                                     
1209          # [B, T, H, M]                                                                                                             
1210          block_mask = parallel_nsa_block_mask(block_indices, block_counts, offsets, block_size)                                     
1211          dk = torch.empty(NV, *k.shape, dtype=k.dtype if NV == 1 else torch.float, device=q.device)                                 
1212          dv = torch.empty(v.shape, dtype=v.dtype, device=q.device)                                                                  
1213                                                                                                                                     
1214          grid = (NV, NS, B * H)                                                                                                     
1215          parallel_nsa_bwd_kernel_dkv[grid](                                                                                         
1216              q=q,                                                                                                                   
1217              k=k,                                                                                                                   
1218              v=v,                                                                                                                   
1219              lse=lse,                                                                                                               
1220              delta=delta,                                                                                                           
1221              do=do,                                                                                                                 
1222              dk=dk,                                                                                                                 
1223              dv=dv,                                                                                                                 
1224              block_mask=block_mask,                                                                                                 
1225              offsets=offsets,                                                                                                       
1226              chunk_indices=chunk_indices,                                                                                           
1227              scale=scale,                                                                                                           
1228              T=T,                                                                                                                   
1229              B=B,                                                                                                                   
1230              H=H,                                                                                                                   
1231              HQ=HQ,                                                                                                                 
1232              G=G,                                                                                                                   
1233              K=K,                                                                                                                   
1234              V=V,                                                                                                                   
1235              M=block_mask.shape[-1],                                                                                                
1236              BS=BS,                                                                                                                 
1237              BK=BK,                                                                                                                 
1238              BV=BV                                                                                                                  
1239          )                                                                                                                          
1240          dk = dk.sum(0)                                                                                                             
1241          return dq, dk, dv

1245      class ParallelNSAFunction(torch.autograd.Function):                                                                            
1246                                                                                                                                     
1247          @staticmethod                                                                                                              
1248          @contiguous                                                                                                                
1249          @autocast_custom_fwd                                                                                                       
1250          def forward(ctx, q, k, v, block_indices, block_counts, block_size, scale, offsets):                                        
1251              ctx.dtype = q.dtype                                                                                                    
1252                                                                                                                                     
1253              # 2-d sequence indices denoting the offsets of tokens in each sequence                                                 
1254              # for example, if the passed `offsets` is [0, 2, 6],                                                                   
1255              # then there are 2 and 4 tokens in the 1st and 2nd sequences respectively, and `token_indices` will be                 
1256              # [[0, 0], [0, 1], [1, 0], [1, 1], [1, 2], [1, 3]]                                                                     
1257              token_indices = prepare_token_indices(offsets) if offsets is not None else None                                        
1258                                                                                                                                     
1259              o, lse = parallel_nsa_fwd(                                                                                             
1260                  q=q,                                                                                                               
1261                  k=k,                                                                                                               
1262                  v=v,                                                                                                               
1263                  block_indices=block_indices,                                                                                       
1264                  block_counts=block_counts,                                                                                         
1265                  block_size=block_size,                                                                                             
1266                  scale=scale,                                                                                                       
1267                  offsets=offsets,                                                                                                   
1268                  token_indices=token_indices                                                                                        
1269              )                                                                                                                      
1270              ctx.save_for_backward(q, k, v, o, lse)                                                                                 
1271              ctx.block_indices = block_indices                                                                                      
1272              ctx.block_counts = block_counts                                                                                        
1273              ctx.offsets = offsets                                                                                                  
1274              ctx.token_indices = token_indices                                                                                      
1275              ctx.block_size = block_size                                                                                            
1276              ctx.scale = scale                                                                                                      
1277              return o.to(q.dtype)

1279          @staticmethod                                                                                                              
1280          @contiguous                                                                                                                
1281          @autocast_custom_bwd                                                                                                       
1282          def backward(ctx, do):                                                                                                     
1283              q, k, v, o, lse = ctx.saved_tensors                                                                                    
1284              dq, dk, dv = parallel_nsa_bwd(                                                                                         
1285                  q=q,                                                                                                               
1286                  k=k,                                                                                                               
1287                  v=v,                                                                                                               
1288                  o=o,

1292                  block_counts=ctx.block_counts,                                                                                     
1293                  block_size=ctx.block_size,                                                                                         
1294                  scale=ctx.scale,                                                                                                   
1295                  offsets=ctx.offsets,                                                                                               
1296                  token_indices=ctx.token_indices                                                                                    
1297              )

1319      def parallel_nsa(                                                                                                              
1320          q: torch.Tensor,                                                                                                           
1321          k: torch.Tensor,                                                                                                           
1322          v: torch.Tensor,                                                                                                           
1323          g_cmp: torch.Tensor,                                                                                                       
1324          g_slc: torch.Tensor,                                                                                                       
1325          g_swa: torch.Tensor,                                                                                                       
1326          block_indices: Optional[torch.LongTensor] = None,                                                                          
1327          block_counts: Union[torch.LongTensor, int] = 16,                                                                           
1328          block_size: int = 64,                                                                                                      
1329          window_size: int = 0,                                                                                                      
1330          scale: Optional[float] = None,                                                                                             
1331          cu_seqlens: Optional[torch.LongTensor] = None,                                                                             
1332          head_first: bool = False                                                                                                   
1333      ) -> torch.Tensor:                                                                                                             
1334          r"""                                                                                                                       
1335          Args:                                                                                                                      
1336              q (torch.Tensor):                                                                                                      
1337                  queries of shape `[B, T, HQ, K]` if `head_first=False` else `[B, HQ, T, K]`.                                       
1338              k (torch.Tensor):                                                                                                      
1339                  keys of shape `[B, T, H, K]` if `head_first=False` else `[B, H, T, K]`.                                            
1340                  GQA is enforced here. The ratio of query heads (HQ) to key/value heads (H) must be a power of 2 and >=16.          
1341              v (torch.Tensor):                                                                                                      
1342                  values of shape `[B, T, H, V]` if `head_first=False` else `[B, H, T, V]`.                                          
1343              g_cmp (torch.Tensor):                                                                                                  
1344                  Gate score for compressed attention of shape `[B, T, HQ]` if  `head_first=False` else `[B, HQ, T]`.                
1345              g_slc (torch.Tensor):                                                                                                  
1346                  Gate score for selected attention of shape `[B, T, HQ]` if  `head_first=False` else `[B, HQ, T]`.                  
1347              g_swa (torch.Tensor):                                                                                                  
1348                  Gate score for sliding attentionof shape `[B, T, HQ]` if  `head_first=False` else `[B, HQ, T]`.                    
1349              block_indices (torch.LongTensor):                                                                                      
1350                  Block indices of shape `[B, T, H, S]` if `head_first=False` else `[B, H, T, S]`.                                   
1351                  `S` is the number of selected blocks for each query token, which is set to 16 in the paper.                        
1352                  If `g_cmp` is provided, the passed `block_indices` will be ignored.                                                
1353              block_counts (Optional[Union[torch.LongTensor, int]]):                                                                 
1354                  Number of selected blocks for each query.                                                                          
1355                  If a tensor is provided, with shape `[B, T, H]` if `head_first=False` else `[B, H, T]`,                            
1356                  each query can select the same number of blocks.                                                                   
1357                  If not provided, it will default to 16.                                                                            
1358              block_size (int):                                                                                                      
1359                  Selected block size. Default: 64.                                                                                  
1360              window_size (int):                                                                                                     
1361                  Sliding window size. Default: 0.                                                                                   
1362              scale (Optional[int]):                                                                                                 
1363                  Scale factor for attention scores.                                                                                 
1364                  If not provided, it will default to `1 / sqrt(K)`. Default: `None`.                                                
1365              head_first (Optional[bool]):                                                                                           
1366                  Whether the inputs are in the head-first format. Default: `False`.                                                 
1367              cu_seqlens (torch.LongTensor):                                                                                         
1368                  Cumulative sequence lengths of shape `[N+1]` used for variable-length training,                                    
1369                  consistent with the FlashAttention API.                                                                            
1370                                                                                                                                     
1371          Returns:                                                                                                                   
1372              o (torch.Tensor):                                                                                                      
1373                  Outputs of shape `[B, T, HQ, V]` if `head_first=False` else `[B, HQ, T, V]`.                                       
1374          """                                                                                                                        
1375          assert block_counts is not None, "block counts must be provided for selection"                                             
1376          if scale is None:                                                                                                          
1377              scale = k.shape[-1] ** -0.5                                                                                            
1378          if cu_seqlens is not None:                                                                                                 
1379              assert q.shape[0] == 1, "batch size must be 1 when cu_seqlens are provided"                                            
1380          if head_first:                                                                                                             
1381              q, k, v = map(lambda x: rearrange(x, 'b h t d -> b t h d'), (q, k, v))                                                 
1382              g_cmp, g_slc, g_swa = map(lambda x: rearrange(x, 'b h t -> b t h') if x is not None else None, (g_cmp, g_slc, g_swa))  
1383              if not isinstance(block_counts, int):                                                                                  
1384                  block_counts = rearrange(block_counts, 'b h t -> b t h')                                                           
1385          assert q.shape[2] % (k.shape[2] * 16) == 0, "Group size must be a multiple of 16 in NSA"                                   
1386                                                                                                                                     
1387          k_cmp, v_cmp = mean_pooling(k, block_size, cu_seqlens), mean_pooling(v, block_size, cu_seqlens)                            
1388          o_cmp, lse_cmp = None, None                                                                                                
1389          if g_cmp is not None:                                                                                                      
1390              o_cmp, lse_cmp = parallel_nsa_compression(                                                                             
1391                  q=q,                                                                                                               
1392                  k=k_cmp,                                                                                                           
1393                  v=v_cmp,                                                                                                           
1394                  block_size=block_size,                                                                                             
1395                  scale=scale,                                                                                                       
1396                  offsets=cu_seqlens                                                                                                 
1397              )                                                                                                                      
1398              if block_indices is not None:                                                                                          
1399                  warnings.warn("`block_indices` will be ignored when `g_cmp` is provided")                                          
1400              block_indices = parallel_nsa_topk(                                                                                     
1401                  q=q,                                                                                                               
1402                  k=k_cmp,                                                                                                           
1403                  lse=lse_cmp,                                                                                                       
1404                  block_counts=block_counts,                                                                                         
1405                  block_size=block_size,                                                                                             
1406                  scale=scale,                                                                                                       
1407                  offsets=cu_seqlens                                                                                                 
1408              )                                                                                                                      
1409          o_slc = ParallelNSAFunction.apply(q, k, v, block_indices, block_counts, block_size, scale, cu_seqlens)                     
1410          o = o_slc * g_slc.unsqueeze(-1)                                                                                            
1411          if o_cmp is not None:                                                                                                      
1412              o = torch.addcmul(o, o_cmp, g_cmp.unsqueeze(-1))                                                                       
1413          if window_size > 0:                                                                                                        
1414              if cu_seqlens is not None:                                                                                             
1415                  max_seqlen = q.shape[1]                                                                                            
1416                  o_swa = flash_attn_varlen_func(                                                                                    
1417                      q.squeeze(0), k.squeeze(0), v.squeeze(0),                                                                      
1418                      cu_seqlens_q=cu_seqlens,                                                                                       
1419                      cu_seqlens_k=cu_seqlens,                                                                                       
1420                      max_seqlen_q=max_seqlen,                                                                                       
1421                      max_seqlen_k=max_seqlen,                                                                                       
1422                      causal=True,                                                                                                   
1423                      window_size=(window_size-1, 0)                                                                                 
1424                  ).unsqueeze(0)                                                                                                     
1425              else:                                                                                                                  
1426                  o_swa = flash_attn_func(                                                                                           
1427                      q, k, v,                                                                                                       
1428                      causal=True,                                                                                                   
1429                      window_size=(window_size-1, 0)                                                                                 
1430                  )                                                                                                                  
1431              o = torch.addcmul(o, o_swa, g_swa.unsqueeze(-1))                                                                       
1432          if head_first:                                                                                                             
1433              o = rearrange(o, 'b t h d -> b h t d')                                                                                 
1434          return o
